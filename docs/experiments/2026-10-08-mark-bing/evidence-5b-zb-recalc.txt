# 由 scratchpad/r/ZB1-8 圖以 PIL 重算 5b 文件引用的數字
import sys, itertools, numpy as np
from PIL import Image
R = sys.argv[1]
def ld(n, k): return np.asarray(Image.open(f"{R}/ZB{n}/1280x720_n1_slot0.{k}.png").convert("RGB")).astype(int)
def nd(a, b): return int((np.abs(a - b).max(axis=2) > 0).sum())
F = {n: ld(n, 'f') for n in range(1, 9)}; O = {n: ld(n, 'o') for n in range(1, 9)}; P = {n: ld(n, '') if False else np.asarray(Image.open(f"{R}/ZB{n}/1280x720_n1_slot0.png").convert("RGB")).astype(int) for n in range(1, 9)}
print('每個基準的足跡（f.png 對 o.png 相異像素）:')
for n in range(1, 9): print(f"  ZB{n}", nd(F[n], O[n]))
print("每個基準 hitstop 全開 f.png 對原截圖 .png 相異像素:")
for n in range(1, 9): print(f"  ZB{n}", nd(F[n], P[n]))
print("整幀(原截圖 .png)兩兩相異像素:")
for a, b in itertools.combinations(range(1, 9), 2): print(f"  ZB{a}-ZB{b}", nd(P[a], P[b]))
M = {n: ((np.abs(F[n] - O[n]).max(axis=2)) > 0) for n in F}
print("足跡遮罩兩兩 XOR px:")
v = []
for a, b in itertools.combinations(range(1, 9), 2): x = int((M[a] ^ M[b]).sum()); v.append(x); print(f"  ZB{a}-ZB{b}", x)
print("min/median/max", min(v), sorted(v)[len(v) // 2], max(v))
